PROGRAM testeCompilador;

VAR
    contador, limite, total, indice, ajuste : INTEGER;
    ativo, valido, menor, maior, iguais, diferentes, dentroFaixa : BOOLEAN;
    mensagem : STRING;

BEGIN
    /* Inicio do programa */

    READ(contador, limite);

    contador = 0;
    limite = 10;
    indice = 1;
    ajuste = +5;
    total = -10 + contador * 2 / 1;

    ativo = TRUE;
    valido = FALSE;

    menor = contador < limite;
    maior = limite > contador;
    iguais = contador == limite;
    diferentes = contador <> limite;
    dentroFaixa = contador <= limite AND limite >= contador;

    valido = ~(contador == limite) AND ativo OR FALSE;

    WRITE("Inicio do teste", contador, limite, total, ativo, valido);

    IF (contador < limite) {
        WRITE("contador menor que limite", contador)
    }
    ELSE {
        WRITE("contador maior ou igual ao limite", contador)
    };

    WHILE (contador < limite) {
        contador = contador + 1;
        total = total + contador;

        IF (contador >= 5) {
            WRITE("contador maior ou igual a cinco", contador)
        }
        ELSE {
            WRITE("contador menor que cinco", contador)
        }
    };

    FOR (indice = 1 TO 3) {
        total = total + indice;
        WRITE("indice do for", indice, total)
    };

    IF (total <> 0) {
        WRITE("total diferente de zero", total);
        WRITE("teste finalizado com sucesso")
    }
    ELSE {
        WRITE("total igual a zero")
    }
END.

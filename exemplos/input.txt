PROGRAM testeProjetoII;

VAR
    valor, limite, soma, i, dobro, metade : INTEGER;
    ativo, passou, parou, maior, menorOuIgual, diferente : BOOLEAN;

BEGIN
    /* Teste completo do compilador com 3AC, otimizacao e assembly */

    READ(valor, limite);

    soma = 0;
    i = 1;
    ativo = TRUE;
    passou = FALSE;
    parou = FALSE;

    dobro = valor * 2;
    metade = limite / 2;

    maior = dobro > metade;
    menorOuIgual = valor <= limite;
    diferente = valor <> limite;

    WRITE("Valores iniciais");
    WRITE(valor, limite, dobro, metade);
    WRITE(maior, menorOuIgual, diferente);

    IF (valor < limite) {
        WRITE("valor menor que limite");
        passou = TRUE
    }
    ELSE {
        WRITE("valor maior ou igual ao limite");
        passou = FALSE
    };

    WHILE (i <= limite) {
        soma = soma + i;

        IF (soma >= 20) {
            parou = TRUE;
            WRITE("soma chegou ou passou de vinte");
            WRITE(soma)
        };

        i = i + 1
    };

    FOR (i = 1 TO 4) {
        dobro = i * 2;
        WRITE("dobro do indice");
        WRITE(i, dobro)
    };

    ativo = passou AND menorOuIgual OR ~diferente;

    IF (ativo) {
        WRITE("programa terminou com estado ativo");
        WRITE(ativo)
    }
    ELSE {
        WRITE("programa terminou com estado inativo");
        WRITE(ativo)
    };

    WRITE("resultado final");
    WRITE(soma, passou, parou)
END.
